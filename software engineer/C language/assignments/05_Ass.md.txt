01). IPL Fan Bot (if-else-if ladder):-


#include <stdio.h>
#include <string.h>

void main(){
    char team[10];
    printf("enter your favorite IPL team (MI,RCB,RR):-");
    scanf("%s",&team);

    if (strcmp(team,"MI")==0){
        printf("Duniya Hila Denge! Go Mumbai Indians!\n");
    }
    else if (strcmp(team,"RCB")==0){
        printf("Ee Sala cup Namde! Royal Challengers Banglore!\n");
    }
    else if(strcmp(team,"RR")==0){
        printf("Halla Bol! Rajsthan Royal!\n");
    }
    else{

    }
}

03). Flipkart Discount Calculator(Nested if statements):-

  #include <stdio.h>

void main(){
    float cartAmount,finalAmount;
    printf("ENter total cart amount:");
    scanf("%f",&cartAmount);

    if(cartAmount>1000){
        if(cartAmount>2000){
            //20% discount for amount>2000
            finalAmount = cartAmount-(cartAmount*0.20);
            printf("20%% discount applied.\n");
        }
        else{
            //10% discount for amount>1000 and <=2000
            finalAmount = cartAmount-(cartAmount*0.10);
            printf("10%% discount applied.\n");
        }
    } else{
        //no discount
        finalAmount = cartAmount;
        printf("No discount applied.\n");
    }
    printf("final amount to pay: %.2f\n",finalAmount);

}
