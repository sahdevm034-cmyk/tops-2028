01). Function to get user initials in uppercase:-


#include <stdio.h>
#include <ctype.h>
#include <string.h>

 void getUserInitials(char name[], char initials[]){
    int i=0,j=0;
    if(name[0]!='\0' && name[0]!=' '){
        initials[j++] = toupper(name[0]);
    }
    for(i=0; name[i]!= '\0'; i++){
        if (name[i] == ' ' && name[i+1] != '\0' && name[i+1]!=' '){
            initials[j++] = toupper(name[i +1]);
        }
 
      }
         initials[j] = '\0';
    }
    void main(){
        char cricketerName[] ="Hardik pandya";
        char initials[10];
        getUserInitials (cricketerName,initials);
        printf("Name: %s\n",cricketerName);
        printf("initials: %s\n",initials);

    }

03).Pass-by-value vs pass-by-Reference(follower count):-

#include <stdio.h>

void increaseFollowerByValue(int count){
    count = count +1000;
    printf("Inside increaseFollowerByValue: %d\n",count);
}
void increaseFollowerByReference(int *count){
    *count = *count +1000;
    printf("inside increaseFollowerByReference:%d\n",count);
}
void main(){
    int followers = 5000;
    printf("original followers: %d\n\n",followers);

    increaseFollowerByValue(followers);
    printf("After increaseFollowerByValue,followers in main: %d\n\n",followers);

    increaseFollowerByReference(&followers);
    printf("After increaseFollowerByReference,followers in main: %d\n",followers);

}

05). Reusable function to  capitalize the first letter of any string:-

#include <stdio.h>
#include <ctype.h>

void capitalizeFirstLetter(char str[]){
    if(str[0]!= '\0'){
        str[0] = toupper(str[0]);
    }
}
void main(){
    char productName[] = "laptop bag";
    char username[] = "sahdev_mali";

    capitalizeFirstLetter(productName);
    capitalizeFirstLetter(username);

    printf("capitalized product Name: %s\n",productName);
printf("capitalized username: %s\n",username);
