01). 1D array dailysteps for 7 days(loop traversal):-

#include <stdio.h>

void main(){
    int dailySteps[7] = {4500,6200,5800,7100,8300,9500,6000};
    int i;
    printf("Daily step count for the week:\n");
    for (i=0;i<7;i++){
        printf("Day %d: %d steps\n",i + 1,dailySteps[i]);
    }
}

03). Function to calculate average spend from 7-day Zomato orders :- 

  
   #include <stdio.h>

float calculateAveragespend(int orders[],int size){
    int sum = 0;
    int i;

    for(i=0;i<size;i++){
        sum += orders[i];
    }
    return (float)sum/size;
}
void main(){
    int zomatoOrders[7] = {250,320,180,450,600,350,290};
    float avg = calculateAveragespend(zomatoOrders,7);

    printf("total days: 7\n");
    printf("Average daily zomato spend: Rs. %.2f\n",avg);
}
