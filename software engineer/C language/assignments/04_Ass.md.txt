02). Filpkart-style discount calculator :-


#include <stdio.h>

void main(){
    float productprice = 999.00;
    float discountpercent = 10.0;
    int ismember =1;
 
    float discountamount = (productprice * discountpercent)/100;
    float priceafterdiscount =productprice -discountamount;

    if(ismember){
        float extradiscount =(priceafterdiscount*5)/100;
        priceafterdiscount =priceafterdiscount - extradiscount;
    }
    printf("initial price:-%.2f\n",productprice);
    printf("final price after discount:-%.2f\n",priceafterdiscount);

}

05).pre-increment vs post-increment:-


#include <stdio.h>

void main(){
    int followercount = 199;
    printf("initial followercount:- %d\n",followercount);
    //post-increment:-(count++)
    printf("during followercount:-%d\n",followercount++);
    printf("after followercount:- %d\n",followercount);

    //pre-increment:-(++count)
    printf("during followercount:-%d\n",++followercount);
    printf("after followercount:- %d\n",followercount);
}
