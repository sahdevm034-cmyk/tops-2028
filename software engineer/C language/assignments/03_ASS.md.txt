02).Calculate final price of  Zomato order with GST:-



#include <stdio.h>

void main(){
    float GST_RATE = 0.18;
    float baseprice = 299.00;
    float finalprice = baseprice+(baseprice*GST_RATE);

    printf("base price: %.2f\n",baseprice);
    printf("final price with GST: %.2f\n",finalprice);
}
