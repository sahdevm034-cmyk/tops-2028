02). Open playlist.txt in read mode ("r") and display each song on console:-


#include <stdio.h>

void main(){
    FILE *file = fopen("playlist.txt","r");
    char song[100];

    if(file == NULL){
        printf("Error opening file!\n");
    }
    printf("songs in playlist.txt:\n");
    while(fgets(song,sizeof(song),file) != NULL){
        printf("%s",song);
    }
    fclose(file);
}

04). Read songs from playlist.txt and print only those containing  the word 'love' (case-insensitive):-



#include <stdio.h>
#include <string.h>
#include <ctype.h>

// Helper function to check if word contains 'love' in case-insenstive way
int containsLove(const char *str){
    char lowerstr[100];
    int i;
    for (i =0;str[i] !='\0'; i++){
        lowerstr[i] = tolower(str[i]);
    }
    lowerstr[i] ='\0';

    return strstr(lowerstr,"love") !=NULL;

}
void main(){
    FILE *file = fopen("playlist.txt","r");

    }
