01). 5X5 Grid pattern representing Instagram post feed :- 

#include <stdio.h>

void main(){
     int  i,j;   // i=rows, j=colum
     for (i=1;i<=5;i++){
        for(j=1;j<=5;j++){
            printf("[📷]");
        }
        printf("\n");
     }
}

03).Centered star pyramid pattern with 6 rows:-

#include <stdio.h>

void main(){
    int i,j,space;
    int rows = 6;
    for(i=1;i<=rows;i++){
        for(space=1;space<= rows - i;space++){
            printf(" ");
        }
        for(j =1;j <=(2*i-1);j++){
            printf("*");
        }
        printf("\n");
    }
}


04).4X4 Grid pattern of alternating 0s and 1s:-


#include <stdio.h>

void main(){
    int i,j;
    for(i=1;i<=4;i++){
        for(j=1; j<=4;j++){
            if((i+j)% 2==0){
                printf("0 ");
        
            }
            else{
                printf("1 ");
            }
        }
        printf("\n");
    }
}
